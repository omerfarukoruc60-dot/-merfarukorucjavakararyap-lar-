import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        System.out.println("Ucgenin 3 kenar uzunlugunu girin turunu soyleyeyim");
        System.out.println("birinci kenar:");
        int kenar1 = input.nextInt();
        System.out.println("ikinci kenar:");
        int kenar2 = input.nextInt();
        System.out.println("ucuncu kenar:");
        int kenar3 = input.nextInt();
        if (kenar1 <= 0 || kenar2 <= 0 || kenar3 <= 0) {
            System.out.println("kenar uzunlugu 0 veya negatif olamaz");
        } else if (kenar1 + kenar2 > kenar3 && kenar1 + kenar3 > kenar2 && kenar2 + kenar3 > kenar1) {
            if (kenar1 == kenar2 && kenar2 == kenar3) {
                System.out.println("ucgeniniz eskenar");
            } else if (kenar1 == kenar2 || kenar1 == kenar3 || kenar2 == kenar3) {
                System.out.println("ucgeniniz ikizkenar");
            } else {
                System.out.println("ucgeniniz cesitkenar");
            }
        } else {
            System.out.println("bu kenarlarla ucgen olusmaz maalesef");
        }
    }
}
