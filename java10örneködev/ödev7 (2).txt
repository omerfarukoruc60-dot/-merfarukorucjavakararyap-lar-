import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        System.out.println("Bir sayi giriniz bolunebilme durumunu soyleyeyim");
        int sayi = input.nextInt();
        if (sayi % 3 == 0 && sayi % 5 == 0) {
            System.out.println("sayiniz " + sayi + " hem 3'e hem 5'e bolunur");
        } else if (sayi % 3 == 0) {
            System.out.println("sayiniz " + sayi + " sadece 3'e bolunur");
        } else if (sayi % 5 == 0) {
            System.out.println("sayiniz " + sayi + " sadece 5'e bolunur");
        } else {
            System.out.println("sayiniz " + sayi + " ne 3'e ne 5'e bolunur");
        }
    }
}
